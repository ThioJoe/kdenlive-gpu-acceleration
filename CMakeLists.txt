# GPU acceleration for Kdenlive: gpufx, an MLT module running some MLT filters
# on NVIDIA GPUs with CUDA. See README.md.
#
#   cmake -S . -B build
#   cmake --build build
#   cmake --install build --prefix <Kdenlive's folder>
#
# MLT is found through its CMake package (Mlt7). An MLT without development
# files, like the one in Kdenlive's Windows standalone build, can be given as
# headers (the src directory of MLT's source) and its library instead:
#
#   -DMLT_INCLUDE_DIR=<MLT source>/src -DMLT_LIBRARY=<Kdenlive>/bin/libmlt-7.dll
#
# MinGW links DLLs directly, no import library is needed. For Kdenlive's
# Linux AppImage, extracted with --appimage-extract, the library is
# <extracted AppImage>/usr/lib/libmlt-7.so.7.
#
# Building it needs no CUDA toolkit: the kernels are compiled into PTX that is
# shipped in kernels/gpufx_kernels.ptx, and the NVIDIA driver is loaded at
# runtime. Without a driver the module leaves every filter on the CPU.
#
# After changing kernels/gpufx_kernels.cu, regenerate the PTX with nvcc:
#   cmake --build <builddir> --target gpufx_update_ptx

cmake_minimum_required(VERSION 3.16)
project(kdenlive-gpu-acceleration LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(GNUInstallDirs)
include(CTest)

set(MLT_INCLUDE_DIR "" CACHE PATH "MLT headers (the directory with framework/mlt.h), for an MLT without CMake package")
set(MLT_LIBRARY "" CACHE FILEPATH "MLT library, for an MLT without CMake package")
if(MLT_INCLUDE_DIR AND MLT_LIBRARY)
    set(mlt_include_dirs "${MLT_INCLUDE_DIR}")
    if(NOT EXISTS "${MLT_INCLUDE_DIR}/framework/mlt_export.h")
        # MLT's build generates this header, which is not in its source
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mlt_export/mlt_export.h" [[
/* mlt_export.h as generated by MLT's build, for users of the library */
#ifndef MLT_EXPORT_H
#define MLT_EXPORT_H
#ifdef _WIN32
#define MLT_EXPORT __declspec(dllimport)
#else
#define MLT_EXPORT __attribute__((visibility("default")))
#endif
#define MLT_NO_EXPORT
#define MLT_DEPRECATED __attribute__((__deprecated__))
#define MLT_DEPRECATED_EXPORT MLT_EXPORT MLT_DEPRECATED
#define MLT_DEPRECATED_NO_EXPORT MLT_NO_EXPORT MLT_DEPRECATED
#endif
]])
        list(APPEND mlt_include_dirs "${CMAKE_CURRENT_BINARY_DIR}/mlt_export")
    endif()
    add_library(Mlt7::mlt UNKNOWN IMPORTED)
    set_target_properties(Mlt7::mlt PROPERTIES
        IMPORTED_LOCATION "${MLT_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${mlt_include_dirs}")
else()
    find_package(Mlt7 REQUIRED)
endif()

if(MINGW)
    # Kdenlive's own GCC runtime may be older than the one used here: link it
    # statically, so that the module only needs MLT and system libraries
    option(GPUFX_STATIC_RUNTIME "Link the GCC runtime libraries statically" ON)
    if(GPUFX_STATIC_RUNTIME)
        add_link_options(-static)
        # mingw-w64 from 4b51967 to 5b128fc (MSYS2's October 2026 builds)
        # defines __mingw_fe_dfl_env in libmingw32.a, which GCC passes before
        # libmingwex.a: a DLL that gets fesetenv only from the static
        # winpthreads cannot find it. Asking for it up front pulls it from
        # libmingw32.a; with other versions it is found as before.
        add_link_options(-Wl,--undefined=__mingw_fe_dfl_env)
    endif()
endif()

set(GPUFX_SOURCE_PTX "${CMAKE_CURRENT_SOURCE_DIR}/kernels/gpufx_kernels.ptx")
set(GPUFX_KERNEL_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/kernels/gpufx_kernels.cu"
    "${CMAKE_CURRENT_SOURCE_DIR}/kernels/gpufx_device.h"
)

find_program(GPUFX_NVCC nvcc HINTS ENV CUDA_PATH PATH_SUFFIXES bin)
option(GPUFX_BUILD_PTX "Compile the CUDA kernels with nvcc instead of using the bundled PTX" OFF)

function(gpufx_nvcc_command output)
    set(raw "${CMAKE_CURRENT_BINARY_DIR}/gpufx_kernels.nvcc.ptx")
    set(${output}
        COMMAND "${GPUFX_NVCC}" -ptx -arch=compute_50 -O3 -o "${raw}" "${CMAKE_CURRENT_SOURCE_DIR}/kernels/gpufx_kernels.cu"
        PARENT_SCOPE)
endfunction()

if(GPUFX_NVCC)
    gpufx_nvcc_command(nvcc_command)
    add_custom_target(gpufx_update_ptx
        ${nvcc_command}
        COMMAND "${CMAKE_COMMAND}" "-DINPUT=${CMAKE_CURRENT_BINARY_DIR}/gpufx_kernels.nvcc.ptx" "-DOUTPUT=${GPUFX_SOURCE_PTX}"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/GpufxFinalizePtx.cmake"
        DEPENDS ${GPUFX_KERNEL_SOURCES}
        COMMENT "Regenerating kernels/gpufx_kernels.ptx with nvcc"
        VERBATIM)
endif()

if(GPUFX_BUILD_PTX)
    if(NOT GPUFX_NVCC)
        message(FATAL_ERROR "GPUFX_BUILD_PTX is set but nvcc was not found")
    endif()
    set(GPUFX_PTX "${CMAKE_CURRENT_BINARY_DIR}/gpufx_kernels.ptx")
    gpufx_nvcc_command(nvcc_command)
    add_custom_command(OUTPUT "${GPUFX_PTX}"
        ${nvcc_command}
        COMMAND "${CMAKE_COMMAND}" "-DINPUT=${CMAKE_CURRENT_BINARY_DIR}/gpufx_kernels.nvcc.ptx" "-DOUTPUT=${GPUFX_PTX}"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/GpufxFinalizePtx.cmake"
        DEPENDS ${GPUFX_KERNEL_SOURCES} "${CMAKE_CURRENT_SOURCE_DIR}/cmake/GpufxFinalizePtx.cmake"
        COMMENT "Compiling the gpufx CUDA kernels"
        VERBATIM)
else()
    set(GPUFX_PTX "${GPUFX_SOURCE_PTX}")
endif()

add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/gpufx_ptx.cpp"
    COMMAND "${CMAKE_COMMAND}" "-DINPUT=${GPUFX_PTX}" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/gpufx_ptx.cpp"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/GpufxEmbedPtx.cmake"
    DEPENDS "${GPUFX_PTX}" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/GpufxEmbedPtx.cmake"
    COMMENT "Embedding the gpufx kernels"
    VERBATIM)

add_library(mltgpufx MODULE
    src/bridge.cpp
    src/convert.cpp
    src/cuda_driver.cpp
    src/effect_blur.cpp
    src/effect_color.cpp
    src/effect_dropshadow.cpp
    src/effect_frei0r.cpp
    src/effect_qtblend.cpp
    src/gpu_runtime.cpp
    src/module.cpp
    src/qt_draw.cpp
    src/qt_raster.cpp
    src/settings_file.cpp
    src/sws_port.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/gpufx_ptx.cpp"
)
target_link_libraries(mltgpufx PRIVATE Mlt7::mlt ${CMAKE_DL_LIBS})
set_target_properties(mltgpufx PROPERTIES
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED ON
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON
)
if(NOT MSVC)
    # The kernels' emulation and Qt compatibility rely on exact float math
    target_compile_options(mltgpufx PRIVATE -ffp-contract=off)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # A function missing from the libraries it is linked to fails the build
    # instead of the loading, as on Windows
    target_link_options(mltgpufx PRIVATE -Wl,--no-undefined)
endif()

# MLT only loads modules from its own module directory: lib/mlt in Kdenlive's
# Windows builds, usr/lib/mlt-7 in its AppImage (-DGPUFX_MLT_MODULE_DIR=lib/mlt-7
# with the prefix <extracted AppImage>/usr), the moduledir of MLT's pkg-config
# file elsewhere.
set(GPUFX_MLT_MODULE_DIR "" CACHE STRING "Where to install the gpufx MLT module, absolute or relative to the install prefix (defaults to MLT's module directory)")
set(gpufx_install_dir "${GPUFX_MLT_MODULE_DIR}")
if(NOT gpufx_install_dir)
    if(WIN32)
        set(gpufx_install_dir "${CMAKE_INSTALL_LIBDIR}/mlt")
    else()
        find_package(PkgConfig QUIET)
        if(PKG_CONFIG_FOUND)
            pkg_get_variable(gpufx_mlt_moduledir mlt-framework-7 moduledir)
        endif()
        get_filename_component(gpufx_prefix "${CMAKE_INSTALL_PREFIX}" ABSOLUTE)
        if(gpufx_mlt_moduledir AND gpufx_mlt_moduledir MATCHES "^${gpufx_prefix}/")
            set(gpufx_install_dir "${gpufx_mlt_moduledir}")
        else()
            set(gpufx_install_dir "${CMAKE_INSTALL_LIBDIR}/mlt-7")
            if(gpufx_mlt_moduledir)
                message(STATUS "gpufx: MLT's modules are in ${gpufx_mlt_moduledir}, outside of the install prefix. "
                               "The module goes to ${CMAKE_INSTALL_PREFIX}/${gpufx_install_dir}: MLT only loads it "
                               "if it is copied to ${gpufx_mlt_moduledir} (or set GPUFX_MLT_MODULE_DIR).")
            endif()
        endif()
    endif()
endif()
install(TARGETS mltgpufx
    LIBRARY DESTINATION "${gpufx_install_dir}"
    RUNTIME DESTINATION "${gpufx_install_dir}")

option(GPUFX_SETTINGS_PAGE "Build the module's page in Kdenlive's settings dialog (needs Qt 6 and KDE Frameworks 6)" ON)
if(GPUFX_SETTINGS_PAGE)
    find_package(Qt6 QUIET COMPONENTS Core Gui Widgets)
    find_package(KF6Config QUIET)
    find_package(KF6ConfigWidgets QUIET)
    if(Qt6Widgets_FOUND AND KF6Config_FOUND AND KF6ConfigWidgets_FOUND)
        add_subdirectory(settings)
    else()
        message(WARNING "Qt 6 or KDE Frameworks 6 (KConfig, KConfigWidgets) not found: the settings page is not built. "
                        "Set GPUFX_SETTINGS_PAGE to OFF to build the module alone.")
    endif()
endif()

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
